CREATE TABLE public.bets (

  id uuid NOT NULL DEFAULT gen_random_uuid(),

  user_id uuid NOT NULL,

  match_id bigint NOT NULL,

  home_score bigint,

  away_score bigint,

  first_goalscorer_id bigint,

  points_awarded bigint,

  created_at timestamp with time zone NOT NULL DEFAULT now(),

  updated_at timestamp with time zone,

  is_goalless boolean DEFAULT false,

  is_boosted boolean DEFAULT false,

  CONSTRAINT bets_pkey PRIMARY KEY (id),

  CONSTRAINT bets_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id),

  CONSTRAINT bets_match_id_fkey FOREIGN KEY (match_id) REFERENCES public.matches(id),

  CONSTRAINT bets_first_goalscorer_id_fkey FOREIGN KEY (first_goalscorer_id) REFERENCES public.players(id)

);
